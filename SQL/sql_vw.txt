CREATE VIEW [dbo].[agm_bill_cal]
AS
SELECT        chargewise_count.charges, Sum(chargewise_count.amount) AS total, chargewise_count.gen_date, chargewise_count.society_id
FROM            (SELECT        mca.charges, (mca.amount / Count(flat_master.flat_id)) * mc.bill_count AS amount, mc.gen_date, mc.society_id
                          FROM            (SELECT        Count(*) AS bill_count, gen_date, bill_id, society_id
                                                    FROM            dbo.maintenance_cal
                                                    WHERE        status =3 or status=5
                                                    GROUP BY gen_date, society_id, bill_id) AS mc LEFT JOIN
                                                        (SELECT        bills.value AS charge_id, society_id, bill_id
                                                          FROM            dbo.bill_charges AS bc CROSS apply String_split(bc.bill_charges_id, ',') AS bills) AS split_bills ON mc.bill_id = split_bills.bill_id AND mc.society_id = split_bills.society_id LEFT JOIN
                                                    maintenance_charges_audit AS mca ON mca.audit_id = split_bills.charge_id AND mca.society_id = split_bills.society_id LEFT JOIN
                                                    flat_master ON flat_master.society_id = mca.society_id
                          GROUP BY mca.charges, mca.amount, mc.gen_date, mc.society_id, mc.bill_count) AS chargewise_count
GROUP BY chargewise_count.charges, chargewise_count.gen_date, chargewise_count.society_id
GO

CREATE VIEW [dbo].[maintainance]
AS
SELECT        DATEPART(mm, dbo.maintenance_cal.m_date) AS month, DATENAME(mm, DATEADD(mm, DATEPART(mm, dbo.maintenance_cal.m_date), - 1)) AS month_name, DATEPART(yyyy, dbo.maintenance_cal.m_date) AS year, 
                         dbo.owner_search_vw.name AS owner_name, dbo.owner_search_vw.flat_no, dbo.owner_search_vw.w_name, dbo.maintenance_cal.bill_no, dbo.maintenance_cal.total_amount, dbo.owner_search_vw.flat_id, 
                         receivable.receivable_amt, dbo.owner_search_vw.sq_ft, dbo.maintenance_cal.amt_forward, dbo.maintenance_cal.tax_interest_amt, dbo.owner_search_vw.Unit, dbo.owner_search_vw.type, dbo.owner_search_vw.pre_mob, 
                         dbo.owner_search_vw.bed, dbo.owner_search_vw.email, dbo.maintenance_cal.advance, dbo.terms.terms, dbo.maintenance_cal.due, dbo.owner_search_vw.build_id, dbo.owner_search_vw.owner_id, 
                         CASE WHEN dbo.maintenance_cal.bill_type = 0 THEN 'Regular' ELSE 'Add-on' END AS bill_type, CASE WHEN dbo.maintenance_cal.bill_type = 0 THEN regular_maint.regular_maintenance ELSE NULL END AS col30_amount, 
                         CASE WHEN dbo.maintenance_cal.bill_type = 0 THEN ' Maintenance Charge' ELSE NULL END AS col30_name, CASE WHEN dbo.maintenance_cal.bill_type = 0 THEN 'Parking Charges' ELSE NULL END AS col29_name, 
                         CASE WHEN dbo.maintenance_cal.bill_type = 0 THEN ISNULL(parking.total_amount, 0) ELSE NULL END AS col29_amount, dbo.owner_search_vw.build_name, dbo.owner_search_vw.society_name, 
                         dbo.building_master.address1, dbo.building_master.address2, dbo.building_master.print_name, dbo.building_master.registration_no, dbo.maintenance_cal.bill_id, dbo.maintenance_cal.society_id, 
                         dbo.owner_search_vw.active_status, dbo.maintenance_cal.gen_date, dbo.maintenance_cal.bill_type AS regular_bill
FROM            (SELECT        owner.owner_id, owner.society_id, owner.sq_ft * account_setting_1.rate_per_sqfeet AS regular_maintenance
                          FROM            dbo.account_setting AS account_setting_1 INNER JOIN
                                                    dbo.owner_search_vw AS owner ON account_setting_1.society_id = owner.society_id) AS regular_maint RIGHT OUTER JOIN
                             (SELECT        SUM(ISNULL(regular_amt, 0)) AS amt_received, owner_id
                               FROM            dbo.receipt
                               WHERE        (date < DATEADD(month, DATEDIFF(month, 0, GETDATE()), 0))
                               GROUP BY owner_id) AS rec_amt RIGHT OUTER JOIN
                             (SELECT        SUM(ISNULL(total_amount, 0)) AS receivable_amt, flat_id
                               FROM            dbo.maintenance_cal AS m
                               WHERE        (m_date < DATEADD(month, DATEDIFF(month, 0, GETDATE()), 0))
                               GROUP BY flat_id) AS receivable RIGHT OUTER JOIN
                         dbo.maintenance_cal INNER JOIN
                         dbo.terms INNER JOIN
                         dbo.owner_search_vw ON dbo.terms.society_id = dbo.owner_search_vw.society_id ON dbo.maintenance_cal.flat_id = dbo.owner_search_vw.flat_id INNER JOIN
                         dbo.building_master ON dbo.owner_search_vw.build_id = dbo.building_master.build_id ON receivable.flat_id = dbo.owner_search_vw.owner_id ON rec_amt.owner_id = dbo.owner_search_vw.owner_id ON 
                         regular_maint.owner_id = dbo.owner_search_vw.owner_id LEFT OUTER JOIN
                             (SELECT        o.flat_id, v.society_id, COUNT(v.vehicle_id) AS total_vehicles, SUM(CASE WHEN v.vehicle_type = 0 THEN a.two_wheeler_rate WHEN v.vehicle_type = 1 THEN a.four_wheeler_rate ELSE 0 END) 
                                                         AS total_amount
                               FROM            dbo.Vehicle AS v INNER JOIN
                                                         dbo.owner_search_vw AS o ON o.flat_id = v.flat_id INNER JOIN
                                                         dbo.account_setting AS a ON v.society_id = a.society_id
                               GROUP BY o.flat_id, v.society_id) AS parking ON dbo.owner_search_vw.flat_id = parking.flat_id

                  dbo.Staff_Master AS sm ON vp.vendor_id = sm.staff_id AND vb.service_type = 0 LEFT OUTER JOIN

 CREATE view  [dbo].[vendor_payment] as
 SELECT 
    vm.vendor_name,
    vp.payment_date,
    vp.cheque_no,
    vp.pay_mode,
    vp.society_id,
    vp.bank_name,
    ISNULL(vp.pay_mode + ': ' + vp.transaction_ref, vp.pay_mode + ': ' + vp.cheque_no) AS transaction_ref,
    vb.bill_id,
    vb.bill_number,
    vb.service_type,
    vb.status AS bill_status_id,
    bs.bill_status,
    vp.payment_no,
    vp.payment_id,
    vp.paid_amount,
    vb.due,
	vb.status,
    BillData.amount
FROM dbo.vendor_bill_payments vp
INNER JOIN dbo.vendor_master vm 
    ON vp.vendor_id = vm.vendor_id
CROSS APPLY (
    SELECT ROW_NUMBER() OVER (ORDER BY (SELECT NULL)) AS rn, value AS bill_id
    FROM STRING_SPLIT(vp.bill_details, ',')
) AS BillIds
CROSS APPLY (
    SELECT ROW_NUMBER() OVER (ORDER BY (SELECT NULL)) AS rn, value AS amount
    FROM STRING_SPLIT(vp.bill_paid_amount, ',')
) AS BillData
INNER JOIN dbo.vendor_bills vb 
    ON vb.bill_id = BillIds.bill_id AND vb.vendor_id = vp.vendor_id
INNER JOIN dbo.bill_status bs 
    ON vb.status = bs.status_id
WHERE BillIds.rn = BillData.rn;
GO

GO

CREATE VIEW [dbo].[vw_vendor_bill_payment_status]
AS
SELECT 
    vb.bill_id,
    vb.bill_number,
    CASE 
        WHEN vb.service_type = 0 THEN sm.name 
        ELSE vm.vendor_name 
    END AS vendor_name,
    vb.service_type,
    vb.total_amount,
    vp.payment_id,
    vp.payment_no,
    vp.paid_amount,
    vp.payment_date,
    vp.cheque_no,
    vp.pay_mode,
    vp.society_id,
    vp.bank_name,
    ISNULL(vp.pay_mode + ': ' + vp.transaction_ref,
           vp.pay_mode + ': ' + vp.cheque_no) AS transaction_ref,
    bs.bill_status,

    -- cumulative paid per bill
    SUM(vp.paid_amount) OVER (
        PARTITION BY vb.bill_id 
        ORDER BY vp.payment_id 
        ROWS UNBOUNDED PRECEDING
    ) AS cumulative_paid,

    -- remaining due
    vb.total_amount - SUM(vp.paid_amount) OVER (
        PARTITION BY vb.bill_id 
        ORDER BY vp.payment_id 
        ROWS UNBOUNDED PRECEDING
    ) AS due,

    -- every payment is paid
    'Paid' AS payment_status

FROM vendor_bills vb
INNER JOIN vendor_bill_payments vp 
    ON vp.bill_details = vb.bill_id
LEFT JOIN Staff_Master sm 
    ON vp.vendor_id = sm.staff_id AND vb.service_type = 0
LEFT JOIN vendor_master vm 
    ON vp.vendor_id = vm.vendor_id AND vb.service_type <> 0
INNER JOIN bill_status bs 
    ON vb.status = bs.status_id;
GO
